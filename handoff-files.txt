package backend.service;

import backend.dto.OrderRequest;
import backend.dto.OrderResponse;
import backend.entity.*;
import backend.enums.MovementType;
import backend.enums.OrderStatus;
import backend.repository.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import java.math.BigDecimal;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;

@Service
@RequiredArgsConstructor
public class OrderService {
    private final SalesOrderRepository orders;
    private final CustomerRepository customers;
    private final ProductRepository products;
    private final StockMovementRepository movements;
    private final UserRepository users;

    @Transactional
    public OrderResponse create(OrderRequest r, String email) {
        Customer customer = customers.findById(r.customerId())
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Customer not found"));
        User user = users.findByEmail(email).orElseThrow();

        // merge duplicate products and sort by id so row locks are always taken in the same order
        Map<Long, Integer> wanted = new TreeMap<>();
        for (OrderRequest.Item i : r.items()) {
            wanted.merge(i.productId(), i.quantity(), Integer::sum);
        }

        SalesOrder order = new SalesOrder();
        order.setCustomer(customer);
        order.setCreatedBy(user);
        orders.save(order);

        BigDecimal total = BigDecimal.ZERO;
        for (Map.Entry<Long, Integer> e : wanted.entrySet()) {
            int qty = e.getValue();
            Product p = products.findByIdForUpdate(e.getKey())
                    .filter(Product::isActive)
                    .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND,
                            "Product not found: " + e.getKey()));
            int prev = p.getCurrentStock();
            if (prev < qty)
                throw new ResponseStatusException(HttpStatus.BAD_REQUEST,
                        "Insufficient stock for " + p.getName() + ". Available: " + prev);
            p.setCurrentStock(prev - qty);

            BigDecimal unitPrice = p.getSellingPrice();
            BigDecimal subtotal = unitPrice.multiply(BigDecimal.valueOf(qty));
            OrderItem item = new OrderItem();
            item.setOrder(order);
            item.setProduct(p);
            item.setQuantity(qty);
            item.setUnitPrice(unitPrice);
            item.setSubtotal(subtotal);
            order.getItems().add(item);
            total = total.add(subtotal);

            movements.save(StockMovement.of(p, user, MovementType.STOCK_OUT, qty, prev, prev - qty,
                    "Order #" + order.getId()));
        }
        order.setTotalAmount(total);
        return OrderResponse.from(order);
    }

    @Transactional
    public OrderResponse cancel(Long id, String email) {
        SalesOrder order = orders.findByIdForUpdate(id)
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Order not found"));
        if (order.getStatus() == OrderStatus.CANCELLED)
            throw new ResponseStatusException(HttpStatus.CONFLICT, "Order is already cancelled");
        User user = users.findByEmail(email).orElseThrow();

        for (OrderItem item : order.getItems()) {
            Product p = products.findByIdForUpdate(item.getProduct().getId()).orElseThrow();
            int prev = p.getCurrentStock();
            int updated = prev + item.getQuantity();
            p.setCurrentStock(updated);
            movements.save(StockMovement.of(p, user, MovementType.STOCK_IN, item.getQuantity(), prev, updated,
                    "Order #" + order.getId() + " cancelled"));
        }
        order.setStatus(OrderStatus.CANCELLED);
        return OrderResponse.from(order);
    }

    @Transactional(readOnly = true)
    public List<OrderResponse> list() {
        return orders.findAllByOrderByCreatedAtDesc().stream().map(OrderResponse::from).toList();
    }

    @Transactional(readOnly = true)
    public OrderResponse get(Long id) {
        return OrderResponse.from(orders.findById(id)
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Order not found")));
    }
}

package backend.service;

import backend.dto.MovementResponse;
import backend.dto.ProductResponse;
import backend.dto.StockRequest;
import backend.entity.Product;
import backend.entity.StockMovement;
import backend.entity.User;
import backend.enums.MovementType;
import backend.repository.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import java.util.List;

@Service
@RequiredArgsConstructor
public class StockService {
    private final ProductRepository products;
    private final StockMovementRepository movements;
    private final UserRepository users;

    @Transactional
    public ProductResponse change(StockRequest r, MovementType type, String email) {
        if (type == MovementType.STOCK_OUT && (r.reason() == null || r.reason().isBlank()))
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Reason is required for stock-out");
        Product p = products.findByIdForUpdate(r.productId())
                .filter(Product::isActive)
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Product not found"));
        int prev = p.getCurrentStock();
        int updated = type == MovementType.STOCK_IN ? prev + r.quantity() : prev - r.quantity();
        if (updated < 0)
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Insufficient stock. Available: " + prev);
        p.setCurrentStock(updated);
        User u = users.findByEmail(email).orElseThrow();
        movements.save(StockMovement.of(p, u, type, r.quantity(), prev, updated, r.reason()));
        return ProductResponse.from(p);
    }

    @Transactional(readOnly = true)
    public List<ProductResponse> lowStock() {
        return products.findLowStock().stream().map(ProductResponse::from).toList();
    }

    @Transactional(readOnly = true)
    public List<MovementResponse> history(Long productId) {
        List<StockMovement> list = productId == null
                ? movements.findTop100ByOrderByCreatedAtDesc()
                : movements.findTop100ByProductIdOrderByCreatedAtDesc(productId);
        return list.stream().map(MovementResponse::from).toList();
    }
}

package backend.service;

import backend.dto.ProductRequest;
import backend.dto.ProductResponse;
import backend.entity.*;
import backend.enums.MovementType;
import backend.repository.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import java.util.List;

@Service
@RequiredArgsConstructor
public class ProductService {
    private final ProductRepository products;
    private final CategoryRepository categories;
    private final StockMovementRepository movements;
    private final UserRepository users;

    @Transactional(readOnly = true)
    public List<ProductResponse> list(String keyword) {
        List<Product> result = (keyword == null || keyword.isBlank())
                ? products.findByActiveTrueOrderByNameAsc()
                : products.search(keyword.trim());
        return result.stream().map(ProductResponse::from).toList();
    }

    @Transactional(readOnly = true)
    public ProductResponse get(Long id) {
        return ProductResponse.from(find(id));
    }

    @Transactional
    public ProductResponse create(ProductRequest r, String email) {
        String sku = r.sku().trim().toUpperCase();
        if (products.existsBySkuIgnoreCase(sku))
            throw new ResponseStatusException(HttpStatus.CONFLICT, "SKU already exists");
        Product p = new Product();
        apply(p, r, sku);
        int opening = r.openingStock() == null ? 0 : r.openingStock();
        p.setCurrentStock(opening);
        products.save(p);
        if (opening > 0) {
            User u = users.findByEmail(email).orElseThrow();
            movements.save(StockMovement.of(p, u, MovementType.STOCK_IN, opening, 0, opening, "Opening stock"));
        }
        return ProductResponse.from(p);
    }

    @Transactional
    public ProductResponse update(Long id, ProductRequest r) {
        Product p = find(id);
        String sku = r.sku().trim().toUpperCase();
        if (products.existsBySkuIgnoreCaseAndIdNot(sku, id))
            throw new ResponseStatusException(HttpStatus.CONFLICT, "SKU already exists");
        apply(p, r, sku);
        return ProductResponse.from(p);
    }

    @Transactional
    public void deactivate(Long id) {
        find(id).setActive(false);
    }

    private void apply(Product p, ProductRequest r, String sku) {
        p.setName(r.name().trim());
        p.setSku(sku);
        p.setDescription(r.description());
        p.setPurchasePrice(r.purchasePrice());
        p.setSellingPrice(r.sellingPrice());
        p.setMinimumStock(r.minimumStock());
        p.setCategory(r.categoryId() == null ? null : categories.findById(r.categoryId())
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Category not found")));
    }

    private Product find(Long id) {
        return products.findById(id)
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Product not found"));
    }
}

package backend.repository;

import backend.entity.Product;
import jakarta.persistence.LockModeType;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Lock;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import java.util.List;
import java.util.Optional;

public interface ProductRepository extends JpaRepository<Product, Long> {
    boolean existsBySkuIgnoreCase(String sku);
    boolean existsBySkuIgnoreCaseAndIdNot(String sku, Long id);
    List<Product> findByActiveTrueOrderByNameAsc();

    @Query("select p from Product p where p.active = true and (lower(p.name) like lower(concat('%', :kw, '%')) or lower(p.sku) like lower(concat('%', :kw, '%'))) order by p.name")
    List<Product> search(@Param("kw") String kw);

    @Query("select p from Product p where p.active = true and p.currentStock <= p.minimumStock order by p.currentStock")
    List<Product> findLowStock();

    @Lock(LockModeType.PESSIMISTIC_WRITE)
    @Query("select p from Product p where p.id = :id")
    Optional<Product> findByIdForUpdate(@Param("id") Long id);

    
    @Query("select count(p) from Product p where p.active = true")
    long countActive();

    @Query("select coalesce(sum(p.currentStock), 0) from Product p where p.active = true")
    long totalStock();

    @Query("select count(p) from Product p where p.active = true and p.currentStock <= p.minimumStock")
    long countLowStock();
}

  }

  const estimate = rows.reduce((sum, r) => {
    const p = productById(r.productId);
    return sum + (p ? Number(p.sellingPrice) * Number(r.quantity || 0) : 0);
  }, 0);

  async function create(e) {
    e.preventDefault();
    setFormError("");
    if (!customerId) {
      setFormError("Select a customer.");
      return;
    }
    const items = rows
      .filter((r) => r.productId !== "")
      .map((r) => ({ productId: Number(r.productId), quantity: Number(r.quantity) }));
    if (items.length === 0) {
      setFormError("Add at least one product.");
      return;
    }
    setSaving(true);
    try {
      const res = await api.post("/orders", { customerId: Number(customerId), items });
      setCreateOpen(false);
      setToast({ text: `Order #${res.data.id} placed · ${money(res.data.totalAmount)}`, undoId: res.data.id });
      setRefreshKey((k) => k + 1);
    } catch (err) {
      setFormError(errText(err));
    } finally {
      setSaving(false);
    }
  }

    async function cancel(o) {
    const ok = await confirm({
      title: "Cancel order",
      message: `Cancel order #${o.id}? The stock will be restored.`,
      confirmText: "Cancel order",
      cancelText: "Keep order",
      danger: true,
    });
    if (!ok) return;
    setError("");
    try {
      const res = await api.post(`/orders/${o.id}/cancel`);
      setToast({ text: `Order #${o.id} cancelled. Stock restored.` });
      if (view && view.id === o.id) setView(res.data);
      setRefreshKey((k) => k + 1);
    } catch (err) {
      setError(err.response?.status === 409 ? "This order is already cancelled." : errText(err));
    }
  }

  return (
    <div>
      <div className="d-flex flex-wrap justify-content-between align-items-center gap-2 mb-3">
        <h2 className="m-0">Orders</h2>
        <button className="btn btn-accent" onClick={openCreate}>+ New order</button>
      </div>

      <div className="d-flex flex-wrap gap-2 mb-3">
        <input
          type="search"
          inputMode="search"
          enterKeyHint="search"
          className="form-control flex-grow-1"
          style={{ minWidth: 200, flexBasis: 240 }}
          placeholder="Search by order no, customer or staff"
          autoComplete="off"
          value={keyword}
          onChange={(e) => setKeyword(e.target.value)}
        />
        <select
          className="form-select form-control"
          style={{ width: "auto" }}
          value={statusFilter}
          onChange={(e) => setStatusFilter(e.target.value)}
        >
          <option value="">All statuses</option>
          <option value="CONFIRMED">Confirmed</option>
          <option value="CANCELLED">Cancelled</option>
        </select>
      </div>

      {error && <div className="alert alert-danger py-2">{error}</div>}
      {loading && <p className="text-secondary">Loading...</p>}
      {!loading && !error && shown.length === 0 && <p className="text-secondary">No orders found.</p>}

      {/* Table: tablets and larger */}
      {shown.length > 0 && (
        <div className="table-responsive d-none d-md-block">
          <table className="table table-dark align-middle">
            <thead>
              <tr>
                <th>Order</th><th>Date</th><th>Customer</th><th>By</th>
                <th>Status</th><th className="text-end">Total</th><th></th>
              </tr>
            </thead>
            <tbody>
              {shown.map((o) => (
                <tr key={o.id}>
                  <td>#{o.id}</td>
                  <td className="text-nowrap">{when(o.createdAt)}</td>
                  <td>{o.customerName}</td>
                  <td>{o.createdBy}</td>
                  <td><StatusBadge status={o.status} /></td>
                  <td className="text-end">{money(o.totalAmount)}</td>
                  <td className="text-end text-nowrap">
                    <button className="btn btn-sm btn-outline-light me-2" onClick={() => setView(o)}>View</button>
                    <button className="btn btn-sm btn-outline-light me-2" onClick={() => billPdf(o)}>Bill PDF</button>
                    {o.status === "CONFIRMED" && (
                      <button className="btn btn-sm btn-outline-danger" onClick={() => cancel(o)}>Cancel</button>
                    )}
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}

      {/* Cards: phones */}
      <div className="d-md-none">
        {shown.map((o) => (
          <div className="product-card" key={o.id}>
            <div className="d-flex justify-content-between">
              <strong>Order #{o.id}</strong>
              <StatusBadge status={o.status} />
            </div>
            <div className="text-secondary small">{when(o.createdAt)}</div>
            <div className="mt-2">{o.customerName}</div>
            <div className="d-flex justify-content-between mt-1">
              <span className="text-secondary small">By {o.createdBy}</span>
              <strong>{money(o.totalAmount)}</strong>
            </div>
            <div className="d-flex gap-2 mt-3">
              <button className="btn btn-sm btn-outline-light flex-fill" onClick={() => setView(o)}>View</button>
              <button className="btn btn-sm btn-outline-light flex-fill" onClick={() => billPdf(o)}>Bill PDF</button>
              {o.status === "CONFIRMED" && (
                <button className="btn btn-sm btn-outline-danger flex-fill" onClick={() => cancel(o)}>Cancel</button>
              )}
            </div>
          </div>
        ))}
      </div>

      {/* View order */}
      {view && (
        <>
          <div className="modal d-block" tabIndex="-1">
            <div className="modal-dialog modal-dialog-scrollable">
              <div className="modal-content app-modal">
                <div className="modal-header">
                  <h5 className="modal-title">Order #{view.id}</h5>
                  <button type="button" className="btn-close btn-close-white" onClick={() => setView(null)} />
                </div>
                <div className="modal-body">
                  <div className="d-flex justify-content-between mb-2">
                    <span>{view.customerName}</span>
                    <StatusBadge status={view.status} />
                  </div>
                  <div className="text-secondary small mb-3">{when(view.createdAt)} · By {view.createdBy}</div>
                  {view.items.map((l, i) => (
                    <div className="d-flex justify-content-between border-top py-2" style={{ borderColor: "var(--border)" }} key={i}>
                      <div>
                        {l.productName}
                        <div className="text-secondary small">{l.sku} · {l.quantity} × {money(l.unitPrice)}</div>
                      </div>
                      <strong>{money(l.subtotal)}</strong>
                    </div>
                  ))}
                  <div className="d-flex justify-content-between border-top pt-2 mt-1" style={{ borderColor: "var(--border)" }}>
                    <strong>Total</strong>

                  <button className="btn btn-outline-light" onClick={() => setView(null)}>Close</button>
                </div>
              </div>
            </div>
          </div>
          <div className="modal-backdrop show"></div>
        </>
      )}

      {/* New order */}
      {createOpen && (
        <>
          <div className="modal d-block" tabIndex="-1">
            <div className="modal-dialog modal-lg modal-dialog-scrollable">
              <form className="modal-content app-modal" onSubmit={create}>
                <div className="modal-header">
                  <h5 className="modal-title">New order</h5>
                  <button type="button" className="btn-close btn-close-white" onClick={() => setCreateOpen(false)} />
                </div>
                <div className="modal-body" style={{ minHeight: 320 }}>
                  {formError && <div className="alert alert-danger py-2">{formError}</div>}

                  <label className="form-label">Customer</label>
                  <div className="mb-3">
                    <SearchSelect
                      placeholder="Type to find a customer"
                      value={customerId}
                      onChange={setCustomerId}
                      options={customers.map((c) => ({
                        value: c.id,
                        label: c.name + (c.phone ? ` (${c.phone})` : ""),
                      }))}
                    />
                  </div>

                  <label className="form-label">Products</label>
                  {rows.map((r, i) => (
                    <div className="d-flex flex-wrap gap-2 mb-2" key={i}>
                      <div className="flex-grow-1" style={{ flexBasis: 220 }}>
                        <SearchSelect
                          placeholder="Type to find a product"
                          value={r.productId}
                          onChange={(v) => setRow(i, "productId", v)}
                          options={products.map((p) => ({
                            value: p.id,
                            label: `${p.name} (${p.sku}) · ${money(p.sellingPrice)} · stock ${p.currentStock}`,
                            disabled: rows.some((x, idx) => idx !== i && String(x.productId) === String(p.id)),
                          }))}
                        />
                      </div>
                      <input
                        type="number"
                        min="1"
                        className="form-control"
                        style={{ width: 90 }}
                        value={r.quantity}
                        onChange={(e) => setRow(i, "quantity", e.target.value)}
                      />
                      {rows.length > 1 && (
                        <button type="button" className="btn btn-outline-danger" onClick={() => setRows(rows.filter((_, idx) => idx !== i))}>
                          ✕
                        </button>
                      )}
                    </div>
                  ))}
                  <button type="button" className="btn btn-sm btn-outline-light mt-1" onClick={() => setRows([...rows, emptyRow])}>
                    + Add another product
                  </button>

                  <div className="d-flex justify-content-between mt-4">
                    <span className="text-secondary">Estimated total</span>
                    <strong>{money(estimate)}</strong>
                  </div>
                  <div className="text-secondary small">The final price is set by the server.</div>
                </div>
                <div className="modal-footer">
                  <button type="button" className="btn btn-outline-light" onClick={() => setCreateOpen(false)}>Cancel</button>
                  <button type="submit" className="btn btn-accent" disabled={saving}>
                    {saving ? "Placing order..." : "Place order"}
                  </button>
                </div>
              </form>
            </div>
          </div>
          <div className="modal-backdrop show"></div>
        </>
      )}

      {/* Popup with Undo */}
      {toast && (
        <div className="toast-box">
          <span>{toast.text}</span>
          {toast.undoId && (
            <button className="btn btn-sm btn-accent" onClick={undo}>Undo</button>
          )}
          <button className="btn btn-sm btn-outline-light" onClick={() => setToast(null)}>✕</button>
        </div>
      )}
    </div>
  );
}

import android.app.Activity;

public class OrdersFragment extends BaseListFragment<OrderResponse> {
    private final ActivityResultLauncher<Intent> newOrder = registerForActivityResult(
            new ActivityResultContracts.StartActivityForResult(), result -> {
                Intent d = result.getData();
                if (result.getResultCode() == Activity.RESULT_OK && d != null && getView() != null) {
                    final long id = d.getLongExtra("orderId", -1);
                    double total = d.getDoubleExtra("total", 0);
                    Snackbar.make(getView(), "Order #" + id + " placed \u00B7 " + Fmt.money(total),
                                    Snackbar.LENGTH_LONG)
                            .setAction("UNDO", x -> cancel(id, "Order #" + id + " undone. Stock restored."))
                            .show();
                }
            });

    @Override protected String searchHint() { return "Search order no., customer or staff"; }
    @Override protected String actionLabel() { return "New order"; }
    @Override protected String filterLabel() { return "Hide cancelled orders"; }

    @Override
    protected void onAction() {
        newOrder.launch(new Intent(requireContext(), NewOrderActivity.class));
    }

    @Override
    protected Call<List<OrderResponse>> request(String kw) {
        return ApiClient.get(requireContext()).orders();
    }

    @Override
    protected boolean matches(OrderResponse o, String k) {
        if (cbFilter.isChecked() && "CANCELLED".equals(o.status)) return false;
        if (k.isEmpty()) return true;
        String hay = (o.id + " " + o.customerName + " " + o.createdBy).toLowerCase();
        return hay.contains(k);
    }

    @Override
    protected void sort(List<OrderResponse> items) {
        Collections.sort(items, (a, b) -> Long.compare(b.id, a.id));
    }

    @Override
    protected Row toRow(OrderResponse o) {
        Row r = new Row();
        r.title = "#" + o.id + " \u00B7 " + o.customerName;
        r.sub = Fmt.date(o.createdAt);
        r.right = Fmt.money(o.totalAmount);
        r.rightSub = o.status;
        r.warn = "CANCELLED".equals(o.status);
        return r;
    }

    @Override
    protected void onItemClick(final OrderResponse o) {
        StringBuilder sb = new StringBuilder();
        sb.append("Customer: ").append(o.customerName)
                .append("\nStatus: ").append(o.status)
                .append("\nDate: ").append(Fmt.date(o.createdAt))
                .append("\nBy: ").append(o.createdBy).append("\n\n");
        if (o.items != null) {
            for (OrderLine l : o.items) {
                sb.append(l.quantity).append(" \u00D7 ").append(l.productName)
                        .append("   ").append(Fmt.money(l.subtotal)).append("\n");
            }
        }
        sb.append("\nTotal: ").append(Fmt.money(o.totalAmount));
        MaterialAlertDialogBuilder b = new MaterialAlertDialogBuilder(requireContext())
                .setTitle("Order #" + o.id)
                .setMessage(sb.toString())
                .setPositiveButton("Close", null);
        if ("CONFIRMED".equals(o.status)) {
            b.setNegativeButton("Cancel order", (d, w) -> confirmCancel(o));
        }
        b.setNeutralButton("Bill PDF", (d, w) -> exportBill(o));
        b.show();
    }

    private void exportBill(OrderResponse o) {
        PdfExporter.Table t = new PdfExporter.Table();
        t.title = "Bill #" + o.id;
        t.subtitle = "Supermarket bill";
        t.summary.add(new String[]{"Customer", o.customerName == null ? "-" : o.customerName});
        t.summary.add(new String[]{"Date", Fmt.date(o.createdAt)});
        t.summary.add(new String[]{"Served by", o.createdBy == null ? "-" : o.createdBy});
        t.summary.add(new String[]{"Status", o.status == null ? "-" : o.status});
        t.headers = new String[]{"Item", "Qty", "Rate", "Amount"};
        t.weights = new float[]{5f, 1f, 2f, 2.5f};
        t.right = new boolean[]{false, true, true, true};
        if (o.items != null) {
            for (OrderLine l : o.items) {
                t.rows.add(new String[]{l.productName, String.valueOf(l.quantity),
                        Fmt.money(l.unitPrice), Fmt.money(l.subtotal)});
            }
        }
        t.rows.add(new String[]{"TOTAL", "", "", Fmt.money(o.totalAmount)});
        PdfExporter.export(requireActivity(), "bill_order_" + o.id + ".pdf", t);
    }

    private void confirmCancel(final OrderResponse o) {
        new MaterialAlertDialogBuilder(requireContext())
                .setTitle("Cancel order #" + o.id + "?")
                .setMessage("The stock will be put back.")
                .setNegativeButton("Keep order", null)
                .setPositiveButton("Cancel order", (d, w) ->
                        cancel(o.id, "Order #" + o.id + " cancelled. Stock restored."))
                .show();
    }

    private void cancel(long id, final String okMessage) {
        ApiClient.get(requireContext()).cancelOrder(id).enqueue(new Callback<ResponseBody>() {
            @Override
            public void onResponse(Call<ResponseBody> c, Response<ResponseBody> r) {
                if (!isAdded()) return;
                if (r.isSuccessful()) {
                    Toast.makeText(requireContext(), okMessage, Toast.LENGTH_SHORT).show();
                    reload();
                } else if (r.code() == 401) {
                    Session.expire(requireActivity(), true);
                } else {
                    Toast.makeText(requireContext(), ApiError.message(r), Toast.LENGTH_LONG).show();
                }
            }

            @Override
            public void onFailure(Call<ResponseBody> c, Throwable t) {
                if (isAdded()) Toast.makeText(requireContext(), ApiError.NETWORK, Toast.LENGTH_LONG).show();
            }
        });
    }
}

===== android\app\src\main\java\com\bxostock\app\NewOrderActivity.java =====
package com.bxostock.app;

import android.content.Intent;
import android.os.Bundle;
import android.view.View;
import android.widget.AutoCompleteTextView;
import android.widget.LinearLayout;
import android.widget.TextView;

import androidx.appcompat.app.AppCompatActivity;

import com.bxostock.app.api.ApiClient;
import com.bxostock.app.model.Customer;
import com.bxostock.app.model.OrderRequest;
import com.bxostock.app.model.OrderResponse;
import com.bxostock.app.model.Product;
import com.bxostock.app.util.ApiError;
import com.bxostock.app.util.Fmt;
import com.bxostock.app.util.Picker;
import com.bxostock.app.util.Session;
import com.bxostock.app.util.TokenStore;
import com.google.android.material.appbar.MaterialToolbar;
import com.google.android.material.textfield.TextInputEditText;

import java.util.ArrayList;
import java.util.List;

import retrofit2.Call;
import retrofit2.Callback;
import retrofit2.Response;

public class NewOrderActivity extends AppCompatActivity {
    private static class Line {
        Product p;
        int qty;
    }

    private Picker<Customer> customerPicker;
    private Picker<Product> productPicker;
    private TextInputEditText etQty;
    private TextView tvStock, tvEmpty, tvTotal, tvMessage;
    private LinearLayout cartBox;
    private View btnPlace;
    private final List<Line> cart = new ArrayList<>();

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        if (!TokenStore.isLoggedIn(this)) {
            Session.expire(this, false);
            return;
        }
        setContentView(R.layout.activity_new_order);
        ((MaterialToolbar) findViewById(R.id.toolbar)).setNavigationOnClickListener(v -> finish());
        etQty = findViewById(R.id.etQty);
        tvStock = findViewById(R.id.tvStock);
        tvEmpty = findViewById(R.id.tvEmpty);
        tvTotal = findViewById(R.id.tvTotal);
        tvMessage = findViewById(R.id.tvMessage);
        cartBox = findViewById(R.id.cartBox);
        btnPlace = findViewById(R.id.btnPlace);

        customerPicker = new Picker<Customer>((AutoCompleteTextView) findViewById(R.id.acCustomer), null);
        productPicker = new Picker<>((AutoCompleteTextView) findViewById(R.id.acProduct),
                p -> tvStock.setText("In stock: " + p.currentStock + " \u00B7 " + Fmt.money(p.sellingPrice)));

        findViewById(R.id.btnAdd).setOnClickListener(v -> addLine());
        findViewById(R.id.btnClear).setOnClickListener(v -> {
            cart.clear();
            render();
        });
        btnPlace.setOnClickListener(v -> place());
        render();
        loadCustomers();
        loadProducts();
    }

    private void loadCustomers() {
        ApiClient.get(this).customers(null).enqueue(new Callback<List<Customer>>() {
            @Override
            public void onResponse(Call<List<Customer>> c, Response<List<Customer>> r) {
                if (!r.isSuccessful() || r.body() == null) {
                    fail(r);
                    return;
                }
                customerPicker.set(r.body(), cu -> cu.name + (cu.phone == null || cu.phone.isEmpty() ? "" : " (" + cu.phone + ")"));
                if (r.body().isEmpty()) err("No customers yet. Add a customer first.");
            }

            @Override
            public void onFailure(Call<List<Customer>> c, Throwable t) {
                err(ApiError.NETWORK);
            }
        });
    }

    private void loadProducts() {
        ApiClient.get(this).products(null).enqueue(new Callback<List<Product>>() {
            @Override
            public void onResponse(Call<List<Product>> c, Response<List<Product>> r) {
                if (!r.isSuccessful() || r.body() == null) {
                    fail(r);
                    return;
                }
                List<Product> active = new ArrayList<>();
                for (Product p : r.body()) if (p.active) active.add(p);
                productPicker.set(active, p -> p.sku + " \u00B7 " + p.name + " \u00B7 "
                        + Fmt.money(p.sellingPrice) + " \u00B7 stock " + p.currentStock);
            }

            @Override
            public void onFailure(Call<List<Product>> c, Throwable t) {
                err(ApiError.NETWORK);
            }
        });
    }

    private void addLine() {
        tvMessage.setVisibility(View.GONE);
        Product p = productPicker.selected();
        if (p == null) {
            err("Choose a product from the list.");
            return;
        }
        int qty = parse(etQty.getText().toString().trim());
        if (qty < 1) {
            err("Enter a quantity of at least 1.");
            return;
        }
        Line existing = null;
        for (Line l : cart) if (l.p.id == p.id) existing = l;
        int total = qty + (existing == null ? 0 : existing.qty);
        if (total > p.currentStock) {
            err("Only " + p.currentStock + " of " + p.name + " in stock.");
            return;
        }
        if (existing != null) {
            existing.qty = total;
        } else {
            Line l = new Line();
            l.p = p;
            l.qty = qty;
            cart.add(l);
        }
        etQty.setText("");
        productPicker.clear();
        tvStock.setText("");
        render();
    }

    private void render() {
        cartBox.removeAllViews();
        if (cart.isEmpty()) {
            tvEmpty.setVisibility(View.VISIBLE);
            tvTotal.setText("");
            return;
        }
        tvEmpty.setVisibility(View.GONE);
        double sum = 0;
        for (final Line l : cart) {
            View v = getLayoutInflater().inflate(R.layout.item_cart, cartBox, false);
            double sub = l.p.sellingPrice * l.qty;
            sum += sub;
            ((TextView) v.findViewById(R.id.tvName)).setText(l.p.name);
            ((TextView) v.findViewById(R.id.tvDetail)).setText(l.qty + " \u00D7 " + Fmt.money(l.p.sellingPrice));
            ((TextView) v.findViewById(R.id.tvSub)).setText(Fmt.money(sub));
            v.findViewById(R.id.btnRemove).setOnClickListener(x -> {
                cart.remove(l);
                render();
            });
            cartBox.addView(v);
        }
        tvTotal.setText("Total (estimate): " + Fmt.money(sum));
    }

    private void place() {
        tvMessage.setVisibility(View.GONE);
        Customer cu = customerPicker.selected();
        if (cu == null) {
            err("Choose a customer from the list.");
            return;
        }
        if (cart.isEmpty()) {
            err("Add at least one product.");
            return;
        }
        OrderRequest req = new OrderRequest();
        req.customerId = cu.id;
        for (Line l : cart) req.items.add(new OrderRequest.Item(l.p.id, l.qty));
        btnPlace.setEnabled(false);
        ApiClient.get(this).placeOrder(req).enqueue(new Callback<OrderResponse>() {
            @Override
            public void onResponse(Call<OrderResponse> c, Response<OrderResponse> r) {
                btnPlace.setEnabled(true);
                if (r.isSuccessful() && r.body() != null) {
                    Intent data = new Intent();
                    data.putExtra("orderId", r.body().id);
                    data.putExtra("total", r.body().totalAmount);
                    setResult(RESULT_OK, data);
                    finish();
                } else {
                    fail(r);
                }
            }

            @Override
            public void onFailure(Call<OrderResponse> c, Throwable t) {
                btnPlace.setEnabled(true);
                err(ApiError.NETWORK);
            }
        });
    }

    private int parse(String s) {
        try {
            return Integer.parseInt(s);
        } catch (NumberFormatException e) {
            return 0;
        }
    }

    private void fail(Response<?> r) {
        if (r.code() == 401) Session.expire(this, true);
        else err(ApiError.message(r));
    }

    private void err(String msg) {
        tvMessage.setText(msg);
        tvMessage.setVisibility(View.VISIBLE);
    }
